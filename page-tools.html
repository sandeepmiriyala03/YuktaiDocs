Page tools

Page tools

Six tools that check and explain a page from inside the browser: Accessibility Copilot, a QA report with 15 WCAG checks, form review, design review, Ask page and voice commands. Run them on a practice area with real problems, or on this page.

1. A practice area with problems

Insert a small snippet with typical mistakes (a field without a label, an image without alt text, an icon button without a name, low-contrast text), then run each check on it.

Live demo

Nothing inserted yet. The checks below then run on an empty area.

app/QaButton.tsx
"use client"
import { runQaAudit, auditToMarkdown, reviewForms, reviewDesign, createA11yCopilot } from "@yuktishaalaa/yuktai"

export function QaButton() {
  function check() {
    const root = document.querySelector("main") ?? document
    const report = runQaAudit({ root })            // { score, passed, issues: [{ rule, wcag, message, fix }], summary }
    const forms = reviewForms({ root })            // per form: fields, issues, suggestions
    const design = reviewDesign({ root })          // { score, categories, findings, summary }
    console.log(report.summary, auditToMarkdown(report))

    const copilot = createA11yCopilot({ root })
    copilot.scan()
    copilot.applyAllFixes()                        // safe fixes only; copilot.undoAll() reverts them
  }
  return <button onClick={check}>Check this page</button>
}
app.js
const root = document.querySelector("main")
const report = yuktai.runQaAudit({ root })   // or runQaAudit() for the whole page
console.log(report.score, report.summary)
console.log(yuktai.auditToMarkdown(report))
yuktai.reviewForms({ root })
yuktai.reviewDesign({ root, minFontSize: 16 })

const copilot = yuktai.createA11yCopilot({ root })
copilot.scan(); copilot.applyAllFixes(); copilot.undoAll()

app/tools/page.tsx
"use client"
import { AgenticPanel } from "@yuktishaalaa/yuktai"

export default function Tools() {
  // tabs: copilot, qa, forms, design, ask, voice
  return <AgenticPanel defaultTab="qa" voiceLang="te-IN" engine="rust" />
}
tools.html
<div id="tools"></div>
<script>
  yuktai.mountAgenticPanel("#tools", { defaultTab: "qa", voiceLang: "te-IN", engine: "rust" })
</script>

<!-- or a floating 🛠 button on every page -->
<script>window.YUKTAI_CONFIG = { showPageTools: true, defaultTab: "qa" }</script>
